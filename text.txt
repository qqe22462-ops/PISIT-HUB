local function parseAssetId(id)
    if not id or id == "" then return "" end
    local strId = tostring(id)
    if not strId:find("rbxassetid://") then
        return "rbxassetid://" .. strId
    end
    return strId
end

    pcall(function()

        sound.Volume = volume or 1
        sound.Parent = SoundService
        sound:Play()
        sound.Ended:Connect(function()
            sound:Destroy()
        end)
    end)
end

local function saveConfig()
    pcall(function()
        if writefile then
            local dataToSave = {}
            for k, v in pairs(Config) do
                if typeof(v) == "EnumItem" then
                    dataToSave[k] = v.Name
                else
                    dataToSave[k] = v
                end
            end
            writefile(CONFIG_FILE, HttpService:JSONEncode(dataToSave))
        end
    end)
end

                    Config[k] = v
                end
            end
        end
    end)
end

local function makeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function createSlider(parentPage, translationKey, minVal, maxVal, currentVal, isDecimal, callback)
    local frame = Instance.new("Frame")
    frame.BackgroundColor3 = THEME.BgSecondary
    frame.BackgroundTransparency = 0.3
    frame.ZIndex = 2
    frame.Parent = parentPage

    Instance.new("UICorner", frame).CornerRadius = UDim.new(0, 8)
    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(30, 45, 75)
    stroke.Thickness = 1
    stroke.Parent = frame

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -16, 0, 14)
    label.Position = UDim2.new(0, 10, 0, 4)
    label.BackgroundTransparency = 1
    label.Text = Translations[Config.CurrentLang][translationKey] .. ": " .. tostring(currentVal)
    label.TextColor3 = THEME.TextMain
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 11
    label.ZIndex = 2
    label.Parent = frame

    local sliderBg = Instance.new("Frame")
    sliderBg.Size = UDim2.new(1, -20, 0, 6)
    sliderBg.Position = UDim2.new(0, 10, 0, 24)
    sliderBg.BackgroundColor3 = THEME.BgCard
    sliderBg.ZIndex = 2
    sliderBg.Parent = frame
    Instance.new("UICorner", sliderBg).CornerRadius = UDim.new(1, 0)

    local sliderFill = Instance.new("Frame")
    sliderFill.Size = UDim2.new((currentVal - minVal) / (maxVal - minVal), 0, 1, 0)
    sliderFill.BackgroundColor3 = THEME.Accent
    sliderFill.ZIndex = 2
    sliderFill.Parent = sliderBg
    Instance.new("UICorner", sliderFill).CornerRadius = UDim.new(1, 0)

    local sliderBtn = Instance.new("TextButton")
    sliderBtn.Size = UDim2.new(1, 0, 1, 0)
    sliderBtn.BackgroundTransparency = 1
    sliderBtn.Text = ""
    sliderBtn.ZIndex = 2
    sliderBtn.Parent = sliderBg

    local dragging = false
    sliderBtn.MouseButton1Down:Connect(function() 
        dragging = true 
        playSound(Config.ButtonClickSound, 0.5)
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local pos = math.clamp((input.Position.X - sliderBg.AbsolutePosition.X) / sliderBg.AbsoluteSize.X, 0, 1)
            sliderFill.Size = UDim2.new(pos, 0, 1, 0)
            
            local val
            if isDecimal then
                val = math.round((minVal + ((maxVal - minVal) * pos)) * 10) / 10
            else
                val = math.floor(minVal + ((maxVal - minVal) * pos))
            end
            
            label.Text = Translations[Config.CurrentLang][translationKey] .. ": " .. tostring(val)
            callback(val)
        end
    end)

    table.insert(UpdateLanguageElements, function(lang)
        if Translations[lang][translationKey] then
            local currentText = label.Text:match(":%s*([%d%.]+)") or tostring(currentVal)
            label.Text = Translations[lang][translationKey] .. ": " .. currentText
        end
    end)
end

local function createButtonAction(parentPage, translationKey, callback)
    local frame = Instance.new("Frame")
    frame.BackgroundColor3 = THEME.BgSecondary
    frame.BackgroundTransparency = 0.3
    frame.ZIndex = 2
    frame.Parent = parentPage

    Instance.new("UICorner", frame).CornerRadius = UDim.new(0, 8)
    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(30, 45, 75)
    stroke.Thickness = 1
    stroke.Parent = frame

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 1, 0)
    btn.BackgroundTransparency = 1
    btn.Text = Translations[Config.CurrentLang][translationKey]
    btn.TextColor3 = THEME.TextMain
    btn.Font = Enum.Font.GothamMedium
    btn.TextSize = 12
    btn.ZIndex = 2
    btn.Parent = frame

    btn.MouseButton1Click:Connect(function()
        playSound(Config.ButtonClickSound, 1)
        callback()
    end)

    table.insert(UpdateLanguageElements, function(lang)
        if Translations[lang][translationKey] then
            btn.Text = Translations[lang][translationKey]
        end
    end)
    return btn
end

local function findToolInPlayer(player, toolName)
    if not player then return nil end
    local char = player.Character
    local backpack = player:FindFirstChild("Backpack")
    
    if char then
        for _, item in ipairs(char:GetChildren()) do
            if item:IsA("Tool") and item.Name:lower():find(toolName:lower()) then
                return item
            end
        end
    end
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") and item.Name:lower():find(toolName:lower()) then
                return item
            end
        end
    end
    return nil
end

local function detectMM2Role(player)
    if not player then return "Innocent", THEME.Innocent end
    if findToolInPlayer(player, "Knife") then
        return "Murderer", THEME.Murderer
    elseif findToolInPlayer(player, "Gun") then
        return "Sheriff", THEME.Sheriff
    end
    return "Innocent", THEME.Innocent
end

local function isPlayerAlive(player)
    if not player or not player.Character then return false end
    local hum = player.Character:FindFirstChildOfClass("Humanoid")
    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
    return hum and hum.Health > 0 and hrp ~= nil
end

RunService.RenderStepped:Connect(function()
    if Config.AimbotEnabled and isPlayerAlive(LocalPlayer) then
        local myRole, _ = detectMM2Role(LocalPlayer)
        local closestTarget = nil
        local shortestDist = math.huge

        for _, target in ipairs(Players:GetPlayers()) do
            if target ~= LocalPlayer and isPlayerAlive(target) then
                local targetRole, _ = detectMM2Role(target)
                local shouldTarget = false

                if myRole == "Murderer" then
                    if targetRole == "Sheriff" or targetRole == "Innocent" then shouldTarget = true end
                elseif myRole == "Sheriff" or myRole == "Innocent" then
                    if targetRole == "Murderer" then shouldTarget = true end
                end

                if shouldTarget and target.Character:FindFirstChild("Head") then
                    local head = target.Character.Head
                    local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                    if onScreen then
                        local mousePos = UserInputService:GetMouseLocation()
                        local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                        if dist < shortestDist then
                            shortestDist = dist
                            closestTarget = head
                        end
                    end
                end
            end
        end

        if closestTarget then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, closestTarget.Position)
        end
    end
end)

task.spawn(function()
    while task.wait(0.2) do
        if Config.AutoKillMurderer and isPlayerAlive(LocalPlayer) then
            local myRole, _ = detectMM2Role(LocalPlayer)
            if myRole == "Murderer" then
                local knife = findToolInPlayer(LocalPlayer, "Knife")
                if knife then
                    if knife.Parent == LocalPlayer.Backpack then LocalPlayer.Character.Humanoid:EquipTool(knife) end
                    local targets = Players:GetPlayers()
                    table.sort(targets, function(a, b)
                        local aHasGun = findToolInPlayer(a, "Gun") ~= nil
                        local bHasGun = findToolInPlayer(b, "Gun") ~= nil

                    for _, target in ipairs(targets) do
                        if isPlayerAlive(target) and target ~= LocalPlayer then
                            local tRole, _ = detectMM2Role(target)

                                task.wait(0.18)
                            end
                        end
                    end
                end
            end
        end
    end
end)

RunService.RenderStepped:Connect(function()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local highlight = ESPHolder:FindFirstChild("HL_" .. player.Name)
            local billboard = ESPHolder:FindFirstChild("BB_" .. player.Name)
            local boxFrame = ScreenGui:FindFirstChild("Box_" .. player.Name)

            if isPlayerAlive(player) then
                local roleName, roleColor = detectMM2Role(player)

                if highlight then
                    highlight.Adornee = player.Character
                    highlight.Enabled = Config.ESPEnabled and Config.ShowHighlights
                    highlight.FillColor = roleColor
                    highlight.OutlineColor = roleColor
                end

                if billboard then
                    billboard.Adornee = player.Character.Head
                    billboard.Enabled = Config.ESPEnabled
                    local container = billboard:FindFirstChild("Container")
                    if container then
                        local nameLabel = container:FindFirstChild("NameLabel")
                        local roleLabel = container:FindFirstChild("RoleLabel")

                        if nameLabel then nameLabel.Visible = Config.ShowNames end
                        if roleLabel then 
                            roleLabel.Visible = Config.ShowRoles
                            roleLabel.Text = "[" .. roleName .. "]"
                            roleLabel.TextColor3 = roleColor
                        end
                    end
                end

                if boxFrame then
                    if Config.ESPEnabled and Config.ShowBox then
                        local hrp = player.Character.HumanoidRootPart
                        local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                        if onScreen then
                            local headPos = Camera:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
                            local legPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3.5, 0))
                            local height = math.abs(headPos.Y - legPos.Y)
                            local width = height / 1.5

                            boxFrame.Size = UDim2.new(0, width, 0, height)
                            boxFrame.Position = UDim2.new(0, pos.X - width / 2, 0, pos.Y - height / 2)
                            boxFrame.Visible = true

                            local stroke = boxFrame:FindFirstChild("Stroke")
                            if stroke then stroke.Color = roleColor end
                        else
                            boxFrame.Visible = false
                        end
                    else
                        boxFrame.Visible = false
                    end
                end
            else
                if highlight then highlight.Enabled = false end
                if billboard then billboard.Enabled = false end
                if boxFrame then boxFrame.Visible = false end
            end
        end
    end

    if Config.DroppedGunESP then
        local gunDrop = Workspace:FindFirstChild("GunDrop", true) or Workspace:FindFirstChild("GunServer", true)
        if gunDrop and gunDrop:IsA("BasePart") then
            DroppedGunBB.Adornee = gunDrop
            DroppedGunBB.Enabled = true
            
            if isPlayerAlive(LocalPlayer) then
                local dist = math.floor((LocalPlayer.Character.HumanoidRootPart.Position - gunDrop.Position).Magnitude)
                DroppedGunLabel.Text = "🔫 Dropped Gun [" .. tostring(dist) .. "m]"
            else
                DroppedGunLabel.Text = "🔫 Dropped Gun"
            end
        else
            DroppedGunBB.Enabled = false
        end
    else
        DroppedGunBB.Enabled = false
    end
end)
